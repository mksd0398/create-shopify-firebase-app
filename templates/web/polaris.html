      <s-text-field id="name" label="Name"></s-text-field>
      <s-switch id="enabled" label="Enabled"></s-switch>

  </s-page>
</form>

Add product API Fetch Pattern Use apiFetch() for all API calls. It handles session tokens automatically. Test API Call
JavaScript 
// GET request
const data = await apiFetch("/api/shop");

// POST request
await apiFetch("/api/settings", {
  method: "POST",
  body: JSON.stringify({ greeting: "Hello!" }),
});

// Search with query params
const results = await apiFetch(
  "/api/products/search?q=" + encodeURIComponent(term)
);

// Error handling
try {
  const data = await apiFetch("/api/shop");
  showToast("Success!");
} catch (err) {
  showToast("Error: " + err.message, true);
}

// Loading pattern
const loading = document.getElementById("my-loading");
const content = document.getElementById("my-content");
loading.style.display = "";
content.style.display = "none";
try {
  const data = await apiFetch("/api/data");
  // Render data...
  loading.style.display = "none";
  content.style.display = "";
} catch (err) {
  loading.innerHTML = '<s-banner tone="critical">' +
    '<s-text>' + err.message + '</s-text></s-banner>';
}
